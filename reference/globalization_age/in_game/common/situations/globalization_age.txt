globalization_age = {
	monthly_spawn_chance = monthly_spawn_chance_unique
	hint_tag = hint_globalization_age
	
	can_start = {
		current_age = age_6_revolutions
				
		continent:america = {
			any_location_in_continent = {
				population > 0
				owner_from_old_world = yes
			}
		}		
	}
	
	can_end = {
		always = no
	}
	
	visible = {
		trigger_if = {
			limit = {
				culture = { is_culture_native_american = yes }
			}
			any_market_present_in_country = {
				save_temporary_scope_as = temp_market
				any_goods = {
					has_origin_in_old_world = yes

					save_temporary_scope_as = temp_good
					custom_tooltip = globalization_age_old_world_goods_traded_tt
					hidden_trigger = {
						scope:temp_market ?= {
							OR = {
								is_produced_in_market = scope:temp_good
								is_traded_in_market = scope:temp_good
							}
						}
					}
				}
			}
		}
		trigger_else = {
			any_market_present_in_country = {
				save_temporary_scope_as = temp_market
				any_goods = {
					has_origin_in_new_world = yes

					save_temporary_scope_as = temp_good
					custom_tooltip = globalization_age_new_world_goods_traded_tt
					hidden_trigger = {
						scope:temp_market ?= {
							OR = {
								is_produced_in_market = scope:temp_good
								is_traded_in_market = scope:temp_good
							}
						}
					}
				}
			}
		}
	}
	

	on_start = {
		every_country = {	
			limit = {				
				NOT = { has_variable = is_in_globalization_age }
				any_owned_location = {
					continent = continent:america
					population > 0
				}
			}		
			if = {
				limit = {
					culture = { is_culture_native_american = yes }
					any_market_present_in_country = {
						save_temporary_scope_as = temp_market
						any_goods = {
							has_origin_in_old_world = yes

							save_temporary_scope_as = temp_good
							scope:temp_market = {
								OR = {
									is_produced_in_market = scope:temp_good
									is_traded_in_market = scope:temp_good
								}
							}
						}
					}
				}
				set_variable = { name = is_in_globalization_age value = yes }
				trigger_event_non_silently = globalization_age.1
			}
			else_if = {
				limit = {
					NOT = { culture = { is_culture_native_american = yes } }
					any_market_present_in_country = {
						save_temporary_scope_as = temp_market
						any_goods = {
							has_origin_in_new_world = yes

							save_temporary_scope_as = temp_good
							scope:temp_market = {
								OR = {
									is_produced_in_market = scope:temp_good
									is_traded_in_market = scope:temp_good
								}
							}
						}
					}

				}
				set_variable = { name = is_in_globalization_age value = yes }
				trigger_event_non_silently = globalization_age.1
			}	
		}	
	}

	on_monthly = {
		every_country = {	
			limit = {		
				NOT = { has_variable = is_in_globalization_age}
				any_owned_location = {
					continent = continent:america
					population > 0
				}
			}		
			if = {
				limit = {
					culture = { is_culture_native_american = yes }
					any_market_present_in_country = {
						save_temporary_scope_as = temp_market
						any_goods = {
							has_origin_in_old_world = yes
							
							save_temporary_scope_as = temp_good
							scope:temp_market = {
								OR = {
									is_produced_in_market = scope:temp_good
									is_traded_in_market = scope:temp_good
								}
							}
						}
					}
				}
				set_variable = { name = is_in_globalization_age value = yes }
				trigger_event_non_silently = globalization_age.1
			}
			else_if = {
				limit = {
					NOT = {culture = { is_culture_native_american = yes }}
					any_market_present_in_country = {
						save_temporary_scope_as = temp_market
						any_goods = {
							has_origin_in_new_world = yes
							
							save_temporary_scope_as = temp_good
							scope:temp_market = {
								OR = {
									is_produced_in_market = scope:temp_good
									is_traded_in_market = scope:temp_good
								}
							}
						}
					}

				}
				set_variable = { name = is_in_globalization_age value = yes }
				trigger_event_non_silently = globalization_age.1
			}	
		}	
	}

	tooltip = {
	}


	on_ended = {
		every_country = {	
			limit = {				
				has_variable = is_in_globalization_age
			}	
			remove_variable = is_in_globalization_age
		}	
	}
	
	map_color = {
		if = {
			limit = {
				owner ?= { 
					has_variable = is_in_globalization_age
					culture = { is_culture_native_american = yes }
				}
			}
			value = color_nw_country
		}
		else_if = {
			limit = {
				owner ?= { 
					has_variable = is_in_globalization_age
					NOT = {culture = { is_culture_native_american = yes }}
				}
			}
			value = color_ow_country
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}
	
	legend_key = { 
		desc = "new_world_goods"
		color = color_nw_country
	}
	
	legend_key = { 
		desc = "old_world_goods"
		color = color_ow_country
	}
}
