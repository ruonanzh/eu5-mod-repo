srd_set_slave_source = {
	scope = location
	ai_is_valid = { always = no }

	is_shown = {
		owner ?= { is_human = yes }
		owner = {
			NOT = { has_variable = slave_redistribute_source_location }
		}
		any_pop = {
			pop_type = pop_type:slaves
		}
	}

	is_valid = {
		owner ?= { is_human = yes }
		owner = {
			NOT = { has_variable = slave_redistribute_source_location }
		}
		any_pop = {
			pop_type = pop_type:slaves
		}
	}

	effect = {
		save_scope_as = srd_pick
		owner = {
			if = {
				limit = { has_variable = slave_redistribute_target_location }
				remove_variable = slave_redistribute_target_location
			}
			set_variable = {
				name = slave_redistribute_source_location
				value = scope:srd_pick
			}
			set_variable = {
				name = slave_redistribute_mode
				value = 1
			}
		}
	}
}

srd_import_slaves_here = {
	scope = location
	ai_is_valid = { always = no }

	saved_scopes = { srd_source srd_dest }

	is_shown = {
		owner ?= { is_human = yes }
		owner = {
			has_variable = slave_redistribute_source_location
			OR = {
				NOT = { has_variable = slave_redistribute_mode }
				var:slave_redistribute_mode = 1
			}
		}
		scope:srd_source ?= {
			NOT = { this = root }
			owner = root.owner
			any_pop = {
				pop_type = pop_type:slaves
			}
		}
	}

	is_valid = {
		owner ?= { is_human = yes }
		owner = {
			has_variable = slave_redistribute_source_location
			OR = {
				NOT = { has_variable = slave_redistribute_mode }
				var:slave_redistribute_mode = 1
			}
		}
		scope:srd_source ?= {
			NOT = { this = root }
			owner = root.owner
			any_pop = {
				pop_type = pop_type:slaves
			}
		}
	}

	confirm_title = srd_import_slaves_confirm_title
	confirm_text = srd_import_slaves_confirm_desc

	effect = {
		custom_tooltip = {
			text = srd_slaves_transferred_hint
			owner = {
				var:slave_redistribute_source_location = {
					every_pop = {
						limit = {
							pop_type = pop_type:slaves
						}
						transfer_pop = {
							target = scope:srd_dest
							operator = multiply
							value = 1
						}
					}
				}
				remove_variable = slave_redistribute_source_location
				remove_variable = slave_redistribute_mode
			}
		}
	}
}
